Можно проверить тип соединения командой:
mysql --login-path=appuser --defaults-group-suffix=appuser -e "status"
mysql --login-path=root --defaults-group-suffix=root -e "status"

mysql --login-path=appuser --defaults-group-suffix=appuser school_sheduller -e "SELECT 1;"
mysql --login-path=appuser --defaults-group-suffix=appuser school_sheduller -e "SELECT user, host, plugin FROM mysql.user WHERE user = 'appuser';"

mysql --login-path=root --defaults-group-suffix=root school_sheduller -e "SELECT user, host, plugin FROM mysql.user WHERE user = 'appuser';"
mysql --login-path=root --defaults-group-suffix=root school_sheduller -e "SELECT user, host, plugin FROM mysql.user;"
mysql --login-path=root --defaults-group-suffix=root school_sheduller -e "SELECT user, host, plugin, ssl_type FROM mysql.user WHERE user='root';"
mysql --login-path=root --defaults-group-suffix=root school_sheduller -e "SELECT user, host, plugin, ssl_type FROM mysql.user;"
